<script>
    document.getElementById('dataEscala').valueAsDate = new Date();

    const HORARIOS = ["06:00", "07:00", "08:00", "09:00", "10:00", "11:00", "12:00", "13:00", "14:00", "15:00", "16:00", "17:00", "18:00", "19:00", "22:00"];

    const BLOCOS_SETORES = [
        {
            titulo: "🏢 Restaurantes & Pontos de Venda (PDVs)",
            setores: ["RESTAURANTE TOM", "RESTAURANTE VEZZOSO", "RESTAURANTE CAVE", "CAFÉ DA MANHA TOM", "CAFÉ DA MANHA VEZZOSO", "CHURRASQUEIRA TOM", "SPORT PARK", "ACQUA KIDS", "PIZZARIA"]
        },
        {
            titulo: "☕ Eventos, Serviços & Apoio",
            setores: ["BANQUETES", "ROOM SERVICE", "RESTRIÇÃO", "CARREGADOR"]
        },
        {
            titulo: "👨‍🍳 Produção & Pré-Preparo (Cozinha Central)",
            setores: ["COZINHA QUENTE", "GARDE MANGER", "CONFEITARIA", "AÇOUGUE", "LEGUMERIA", "PADARIA MANHÃ", "PADARIA TARDE", "PADARIA MADRUGADA", "CAFÉ DA MANHÃ MADRUGADA"]
        },
        {
            titulo: "🍽️ Alimentação Interna (Refeitório & Time)",
            setores: ["REFEITORIO", "REFEITÓRIO TARDE", "BUFFET TIME", "CAFÉ DA MANHA TIME", "JANTAR TIME", "CEIA TIME", "LANCHE TARDE TIME"]
        },
        {
            titulo: "👔 Setores Especiais",
            setores: ["VEZZOSO"]
        }
    ];

    const CARGOS_CONFIG = [
        { nome: "Cozinheiro Extra", valor: 317.79, divisorPax: 50, minimo: 1 },
        { nome: "Auxiliar / Garçom Extra", valor: 220.00, divisorPax: 35, minimo: 1 },
        { nome: "Steward Extra", valor: 200.00, divisorPax: 80, minimo: 1 },
        { nome: "Líder Extra", valor: 327.73, divisorPax: 0, fixoPadrao: 1 }
    ];

    let estadoSetores = {};
    let totalSetoresCount = 0;
    let dadosUltimoCalculo = null; 

    window.onload = function() {
        renderizarTodosSetores();
    };

    function getPaxTotal() {
        const ad = parseInt(document.getElementById('adultosGeral').value) || 0;
        const cr = parseInt(document.getElementById('criancasGeral').value) || 0;
        return ad + cr;
    }

    function renderizarTodosSetores() {
        const container = document.getElementById('containerSetores');
        container.innerHTML = "";
        totalSetoresCount = 0;

        BLOCOS_SETORES.forEach(bloco => {
            const divGrupo = document.createElement('div');
            divGrupo.className = 'grupo-titulo';
            divGrupo.innerText = bloco.titulo;
            container.appendChild(divGrupo);

            bloco.setores.forEach(setor => {
                totalSetoresCount++;
                estadoSetores[setor] = null;

                const card = document.createElement('div');
                card.className = 'card-setor';
                card.id = `card_${limparId(setor)}`;

                card.innerHTML = `
                    <div class="setor-header">
                        <div class="setor-nome">
                            <div class="status-dot" id="dot_${limparId(setor)}"></div>
                            <span>${setor}</span>
                        </div>
                        <div class="toggle-status">
                            <button class="btn-status op-nao" id="btn_nao_${limparId(setor)}" onclick="definirStatusSetor('${setor}', false)">🔴 Não Operando</button>
                            <button class="btn-status op-sim" id="btn_sim_${limparId(setor)}" onclick="definirStatusSetor('${setor}', true)">🟢 Em Operação</button>
                        </div>
                    </div>
                    <div class="setor-corpo" id="corpo_${limparId(setor)}">
                        ${gerarCamposCargosHTML(setor)}
                    </div>
                `;
                container.appendChild(card);
            });
        });

        atualizarBarraProgresso();
    }

    function limparId(nome) {
        return nome.replace(/[^a-zA-Z0-0]/g, '_');
    }

    function gerarCamposCargosHTML(setor) {
        let html = `<div style="font-size:11px; font-weight:bold; color:var(--text-muted); margin-bottom:10px;">DIMENSIONAMENTO AUTOMÁTICO DE MÃO DE OBRA - ${setor}:</div>`;
        const pax = getPaxTotal();

        CARGOS_CONFIG.forEach((cargo, idx) => {
            let necessarios = 0;
            if (cargo.divisorPax > 0) {
                necessarios = Math.max(cargo.minimo, Math.ceil(pax / cargo.divisorPax));
            } else {
                necessarios = cargo.fixoPadrao || 1;
            }

            html += `
                <div class="grid-cargos" id="grid_${limparId(setor)}_${idx}">
                    <span style="font-weight:600; font-size:12px;">${cargo.nome}</span>
                    <div class="campo">
                        <label>Qtd. Necessária</label>
                        <input type="number" class="input-necessario" value="${necessarios}" readonly>
                    </div>
                    <div class="campo">
                        <label>Fixo Casa</label>
                        <input type="number" class="input-fixo" value="0" min="0" oninput="calcularLinhaAuto('${setor}', ${idx})">
                    </div>
                    <div class="campo">
                        <label>Extra Sugerido</label>
                        <div class="badge-extra badge-calc">+${necessarios}</div>
                    </div>
                    <div class="campo">
                        <label>Horário Entrada</label>
                        <select class="select-hora">
                            ${HORARIOS.map(h => `<option value="${h}" ${h==='07:00'?'selected':''}>${h}</option>`).join('')}
                        </select>
                    </div>
                </div>
            `;
        });

        return html;
    }

    function calcularLinhaAuto(setor, cargoIdx) {
        const idClean = limparId(setor);
        const grid = document.getElementById(`grid_${idClean}_${cargoIdx}`);
        if (!grid) return;

        const necessarios = parseInt(grid.querySelector('.input-necessario').value) || 0;
        const fixo = parseInt(grid.querySelector('.input-fixo').value) || 0;

        const extra = Math.max(0, necessarios - fixo);
        grid.querySelector('.badge-calc').innerText = `+${extra}`;
    }

    function recalcularTodosOsSetores() {
        const pax = getPaxTotal();
        Object.keys(estadoSetores).forEach(setor => {
            const idClean = limparId(setor);
            CARGOS_CONFIG.forEach((cargo, idx) => {
                const grid = document.getElementById(`grid_${idClean}_${idx}`);
                if (grid) {
                    let necessarios = 0;
                    if (cargo.divisorPax > 0) {
                        necessarios = Math.max(cargo.minimo, Math.ceil(pax / cargo.divisorPax));
                    } else {
                        necessarios = cargo.fixoPadrao || 1;
                    }

                    grid.querySelector('.input-necessario').value = necessarios;
                    calcularLinhaAuto(setor, idx);
                }
            });
        });
    }

    function definirStatusSetor(setor, operando) {
        const idClean = limparId(setor);
        const card = document.getElementById(`card_${idClean}`);
        const corpo = document.getElementById(`corpo_${idClean}`);
        const dot = document.getElementById(`dot_${idClean}`);
        const btnSim = document.getElementById(`btn_sim_${idClean}`);
        const btnNao = document.getElementById(`btn_nao_${idClean}`);

        if (operando) {
            estadoSetores[setor] = 'operando';
            card.className = 'card-setor operando';
            corpo.style.display = 'block';
            dot.className = 'status-dot verde';
            btnSim.classList.add('ativo');
            btnNao.classList.remove('ativo');
            
            CARGOS_CONFIG.forEach((_, idx) => calcularLinhaAuto(setor, idx));
        } else {
            estadoSetores[setor] = 'fechado';
            card.className = 'card-setor fechado';
            corpo.style.display = 'none';
            dot.className = 'status-dot vermelho';
            btnNao.classList.add('ativo');
            btnSim.classList.remove('ativo');
        }

        atualizarBarraProgresso();
    }

    function atualizarBarraProgresso() {
        let avaliados = 0;
        Object.keys(estadoSetores).forEach(setor => {
            if (estadoSetores[setor] !== null) avaliados++;
        });

        const porcentagem = Math.round((avaliados / totalSetoresCount) * 100);
        
        const barFill = document.getElementById('barFill');
        const txtProgresso = document.getElementById('txtProgresso');
        const btnCalcular = document.getElementById('btnCalcular');

        barFill.style.width = `${porcentagem}%`;
        txtProgresso.innerText = `${avaliados} / ${totalSetoresCount} Setores Revisados (${porcentagem}%)`;

        if (porcentagem === 100) {
            barFill.style.backgroundColor = 'var(--primary)';
            btnCalcular.disabled = false;
            btnCalcular.innerText = "⚡ Processar Solicitação para o RH (100% Concluído)";
        } else {
            barFill.style.backgroundColor = 'var(--warning)';
            btnCalcular.disabled = true;
            btnCalcular.innerText = `Revise todos os setores para liberar (${porcentagem}%)`;
        }
    }

    function calcularEPolarizar() {
        const dataEscala = document.getElementById('dataEscala').value;
        const corpoDetalhado = document.getElementById('tabelaCorpoDetalhado');
        corpoDetalhado.innerHTML = "";

        let custoTotalGeral = 0;
        let itensDetalhados = [];

        Object.keys(estadoSetores).forEach(setor => {
            if (estadoSetores[setor] === 'operando') {
                const idClean = limparId(setor);
                const card = document.getElementById(`card_${idClean}`);
                const blocosCargos = card.querySelectorAll('.grid-cargos');

                blocosCargos.forEach((bloc, index) => {
                    const cargoInfo = CARGOS_CONFIG[index];
                    const necessarios = parseInt(bloc.querySelector('.input-necessario').value) || 0;
                    const fixo = parseInt(bloc.querySelector('.input-fixo').value) || 0;
                    const hora = bloc.querySelector('.select-hora').value;

                    const extraQtd = Math.max(0, necessarios - fixo);

                    if (extraQtd > 0) {
                        const valorUnit = cargoInfo.valor;
                        const totalItem = extraQtd * valorUnit;
                        custoTotalGeral += totalItem;

                        itensDetalhados.push({
                            setor: setor,
                            cargo: cargoInfo.nome,
                            hora: hora,
                            extraQtd: extraQtd,
                            valorUnit: valorUnit,
                            totalItem: totalItem
                        });
                    }
                });
            }
        });

        dadosUltimoCalculo = {
            dataEscala: dataEscala,
            custoTotalGeral: custoTotalGeral,
            itens: itensDetalhados
        };

        renderizarRelatoriosNaTela(dadosUltimoCalculo);
    }

    function renderizarRelatoriosNaTela(dados) {
        const corpoDetalhado = document.getElementById('tabelaCorpoDetalhado');
        const corpoConsolidado = document.getElementById('tabelaCorpoConsolidado');
        corpoDetalhado.innerHTML = "";
        corpoConsolidado.innerHTML = "";

        let mapaConsolidado = {};

        dados.itens.forEach(item => {
            // Tabela Detalhada
            corpoDetalhado.innerHTML += `
                <tr>
                    <td><strong>${item.setor}</strong></td>
                    <td>${item.cargo}</td>
                    <td style="text-align:center;"><strong>${item.hora}h</strong></td>
                    <td style="text-align:center;"><strong>${item.extraQtd}</strong></td>
                    <td style="text-align:right;">R$ ${item.valorUnit.toFixed(2)}</td>
                    <td style="text-align:right;">R$ ${item.totalItem.toFixed(2)}</td>
                </tr>
            `;

            // Agrupamento Consolidado
            const chave = `${item.cargo}_${item.hora}`;
            if (!mapaConsolidado[chave]) {
                mapaConsolidado[chave] = {
                    nome: item.cargo,
                    hora: item.hora,
                    qtdTotal: 0,
                    valorUnit: item.valorUnit,
                    totalCusto: 0
                };
            }
            mapaConsolidado[chave].qtdTotal += item.extraQtd;
            mapaConsolidado[chave].totalCusto += item.totalItem;
        });

        // Preenche Consolidado
        Object.values(mapaConsolidado).forEach(item => {
            corpoConsolidado.innerHTML += `
                <tr>
                    <td><strong>${item.nome}</strong></td>
                    <td style="text-align:center;"><strong>${item.hora}h</strong></td>
                    <td style="text-align:center;"><span style="background:var(--primary-light); color:var(--primary); padding:4px 10px; border-radius:12px; font-weight:bold;">${item.qtdTotal} Extras</span></td>
                    <td style="text-align:right;">R$ ${item.valorUnit.toFixed(2)}</td>
                    <td style="text-align:right;"><strong>R$ ${item.totalCusto.toFixed(2)}</strong></td>
                </tr>
            `;
        });

        document.getElementById('custoTotalGeral').innerText = `R$ ${dados.custoTotalGeral.toFixed(2)}`;
        document.getElementById('labelDataConsolidada').innerText = `Data: ${dados.dataEscala.split('-').reverse().join('/')}`;
        document.getElementById('resultadoBox').style.display = 'block';
    }

    function salvarNoHistorico() {
        if (!dadosUltimoCalculo || dadosUltimoCalculo.itens.length === 0) {
            alert("Nenhum cálculo disponível para salvar.");
            return;
        }

        const payload = {
            id: Date.now(),
            dataEnvio: new Date().toLocaleDateString('pt-BR') + ' ' + new Date().toLocaleTimeString('pt-BR'),
            dataEscala: dadosUltimoCalculo.dataEscala.split('-').reverse().join('/'),
            totalCusto: `R$ ${dadosUltimoCalculo.custoTotalGeral.toFixed(2)}`,
            resumo: `${dadosUltimoCalculo.itens.length} lançamentos de extras`,
            detalhes: dadosUltimoCalculo
        };

        let historico = JSON.parse(localStorage.getItem('historicoBourbonV5')) || [];
        historico.unshift(payload);
        localStorage.setItem('historicoBourbonV5', JSON.stringify(historico));

        alert("✅ Enviado com sucesso para o RH e salvo no Histórico!");
    }

    function abrirHistorico() {
        const modal = document.getElementById('modalHistorico');
        const tabela = document.getElementById('tabelaHistoricoCorpo');
        let historico = JSON.parse(localStorage.getItem('historicoBourbonV5')) || [];

        tabela.innerHTML = "";

        if(historico.length === 0) {
            tabela.innerHTML = `<tr><td colspan="5" style="text-align:center; padding:15px;">Nenhum histórico gravado.</td></tr>`;
        } else {
            historico.forEach((h, index) => {
                tabela.innerHTML += `
                    <tr>
                        <td>${h.dataEnvio}</td>
                        <td><strong>${h.dataEscala}</strong></td>
                        <td style="color:var(--success); font-weight:bold;">${h.totalCusto}</td>
                        <td>${h.resumo}</td>
                        <td style="text-align:center;">
                            <button class="btn-action-sm" style="color:var(--primary);" onclick="reabrirRegistro(${index})">👁️ Abrir</button>
                            <button class="btn-action-sm" style="color:#0284c7;" onclick="exportarExcelRegistro(${index})">📊 Excel</button>
                        </td>
                    </tr>
                `;
            });
        }
        modal.style.display = "flex";
    }

    function reabrirRegistro(index) {
        let historico = JSON.parse(localStorage.getItem('historicoBourbonV5')) || [];
        const registro = historico[index];
        if (registro && registro.detalhes) {
            dadosUltimoCalculo = registro.detalhes;
            renderizarRelatoriosNaTela(registro.detalhes);
            fecharHistorico();
            document.getElementById('resultadoBox').scrollIntoView({ behavior: 'smooth' });
        } else {
            alert("Registro não contém dados detalhados.");
        }
    }

    function exportarExcelRegistro(index) {
        let historico = JSON.parse(localStorage.getItem('historicoBourbonV5')) || [];
        const registro = historico[index];
        if (!registro || !registro.detalhes) return;

        let csvContent = "data:text/csv;charset=utf-8,Setor;Cargo;Horario;Qtd Extra;Valor Unitario;Custo Total\n";
        
        registro.detalhes.itens.forEach(item => {
            csvContent += `"${item.setor}";"${item.cargo}";"${item.hora}h";${item.extraQtd};"${item.valorUnit.toFixed(2)}";"${item.totalItem.toFixed(2)}"\n`;
        });

        const encodedUri = encodeURI(csvContent);
        const link = document.createElement("a");
        link.setAttribute("href", encodedUri);
        link.setAttribute("download", `Solicitacao_Extras_${registro.dataEscala.replace(/\//g, '-')}.csv`);
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
    }

    function fecharHistorico() {
        document.getElementById('modalHistorico').style.display = "none";
    }
