<script>
    document.getElementById('dataEscala').valueAsDate = new Date();

    window.onload = function() {
        renderizarHistorico();

    function validarTravaBuffet() {
        const qtdAdultos = parseInt(document.getElementById('adultos').value) || 0;
        const qtdCriancas = parseInt(document.getElementById('criancas').value) || 0;
        const total = qtdAdultos + qtdCriancas;
        const buffetSelect = document.getElementById('buffet');
        const travaMensagem = document.getElementById('travaMensagem');

        if (total === 0) {
            buffetSelect.disabled = false;
            travaMensagem.style.display = "none";
        }
        else if (total < 400) {
            buffetSelect.value = "1";
            buffetSelect.disabled = true; 
            travaMensagem.style.color = "#0d9488";
            travaMensagem.innerText = "🔒 Bloqueado: Menos de 400 pessoas exige 1 Ponto de Buffet (Reduzido).";
            travaMensagem.style.display = "block";
        } 
        else {
            buffetSelect.value = "2"; 
            buffetSelect.disabled = true; 
            travaMensagem.style.color = "var(--success)";
            travaMensagem.innerText = "🔒 Automatizado: Acima de 400 pessoas exige obrigatoriamente 2 Pontos de Buffet.";
            travaMensagem.style.display = "block";
        }
    }

    function calcularEscala(salvarNoHistorico = true) {
        const dataInput = document.getElementById('dataEscala').value;
        const periodoSelecionado = document.getElementById('periodo').value;
        const qtdAdultos = parseInt(document.getElementById('adultos').value) || 0;
        const qtdCriancas = parseInt(document.getElementById('criancas').value) || 0;
        
        if (salvarNoHistorico) {
            validarTravaBuffet();
        }
        const pontosBuffet = document.getElementById('buffet').value;

        const fixosCozinha = parseInt(document.getElementById('fixosCozinha').value) || 0;
        const fixosLider = parseInt(document.getElementById('fixosLider').value) || 0;
        const totalHospedes = qtdAdultos + qtdCriancas;

        if (totalHospedes <= 0) {
            alert("Por favor, insira pelo menos um hóspede!");
            return;
        }

        let dataFormatada = "";
        if (dataInput) {
            const partes = dataInput.split('-');
            dataFormatada = `${partes[2]}/${partes[1]}/${partes[0]}`;
        }

        const valorCozinheiro = 317.79;
        const valorLider = 327.73;
        let demandaCozinheirosTotal = 0;
        let demandaLideresTotal = 0;

        if (totalHospedes <= 400) {
            demandaLideresTotal = 0;
            demandaCozinheirosTotal = (pontosBuffet === "1") ? 8 : 10;
        } else {
            demandaCozinheirosTotal = 13;
            demandaLideresTotal = 1;
        }

        const extrasCozinheirosFaltantes = Math.max(0, demandaCozinheirosTotal - fixosCozinha);
        const extrasLideresFaltantes = Math.max(0, demandaLideresTotal - fixosLider);

        let extrasAgrupados = [];
        let custoTotal = 0;
        let horarioCozinheiro = "06h - 14h";
        let horarioLider = "15h - 23h";

        if (periodoSelecionado === "Almoço") {
            horarioCozinheiro = "10h - 18h"; horárioLider = "11h - 19h";
        } else if (periodoSelecionado === "Jantar") {
            horarioCozinheiro = "16h - 00h"; horárioLider = "17h - 01h";
        } else if (periodoSelecionado === "Evento Especial") {
            horarioCozinheiro = "A definir"; horárioLider = "A definir";
        }

        const estacoesPonto1 = [
            "Estação de Omelete", "Estação de Waffle / Churros", "Misto / Croque Monsieur",
            "Suco / Cuscuz", "Estação de Tapioca", "Pães Assados", "Pista Fria 1", 
            "Pista Fria 2", "Pista Quente 1", "Pista Quente 2", "Back 1", "Back 2"
        ];

        const estacoesPonto2 = [
            "Estação de Omelete (Pt 1)", "Estação de Omelete (Pt 2)", "Estação de Waffle (Pt 1)",
            "Estação de Waffle (Pt 2)", "Misto / Croque Monsieur", "Suco / Cuscuz",
            "Estação de Tapioca", "Pães Assados", "Pista Fria 1", "Pista Fria 2",
            "Pista Quente 1", "Pista Quente 2", "Back 1", "Back 2"
        ];

        let listaEstacoesAtivas = [];
        if (periodoSelecionado === "Café da Manhã" || periodoSelecionado === "Café da Manhã TOM") {
            listaEstacoesAtivas = (pontosBuffet === "2") ? estacoesPonto2 : estacoesPonto1;
        } else {
            listaEstacoesAtivas = ["Apoio Produção", "Auxílio Buffet", "Geral"];
        }

        let areasCozinha = [];
        for (let i = 0; i < extrasCozinheirosFaltantes; i++) {
            areasCozinha.push(listaEstacoesAtivas[i] || "Apoio Geral Cozinha");
        }

        if (extrasCozinheirosFaltantes > 0) {
            const custoCozinheiros = extrasCozinheirosFaltantes * valorCozinheiro;
            extrasAgrupados.push({
                funcao: "Cozinheiro Extra", amount: extrasCozinheirosFaltantes, unitario: valorCozinheiro,
                area: areasCozinha.join(" | "), horario: horarioCozinheiro, custoTotalItem: custoCozinheiros
            });
            custoTotal += custoCozinheiros;
        }

        if (extrasLideresFaltantes > 0) {
            const custoLideres = extrasLideresFaltantes * valorLider;
            extrasAgrupados.push({
                funcao: "Líder Extra", amount: extrasLideresFaltantes, unitario: valorLider,
                area: "Supervisão de Fluxo de Produção", horario: horarioLider, custoTotalItem: custoLideres
            });
            custoTotal += custoLideres;
        }

        const tabelaCorpo = document.getElementById('tabelaCorpo');
        tabelaCorpo.innerHTML = ""; 

        if (extrasAgrupados.length === 0) {
            tabelaCorpo.innerHTML = `<tr><td colspan="6" style="text-align: center; color: var(--success); font-weight: bold; padding: 30px;">🎉 Equipe fixa da casa é suficiente! Sem necessidade de contratações.</td></tr>`;
        } else {
            extrasAgrupados.forEach(item => {
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td><strong>${item.funcao}</strong></td>
                    <td style="text-align: center; font-weight: bold; color: var(--primary); font-size: 16px;">${item.amount}x</td>
                    <td style="text-align: right;">R$ ${item.unitario.toLocaleString('pt-BR', {minimumFractionDigits: 2})}</td>
                    <td style="font-size: 13px; color: var(--text-muted); max-width: 250px; line-height: 1.4;">${item.area}</td>
                    <td>${item.horario}</td>
                    <td style="text-align: right; font-weight: 700;">R$ ${item.custoTotalItem.toLocaleString('pt-BR', {minimumFractionDigits: 2})}</td>
                `;
                tabelaCorpo.appendChild(tr);
            });
        }

        document.getElementById('resumoPeriodo').innerText = periodoSelecionado;
        document.getElementById('resumoPublico').innerText = `${totalHospedes} (${qtdAdultos} Ad / ${qtdCriancas} Cr)`;
        document.getElementById('resumoDemanda').innerText = `${demandaCozinheirosTotal} Coz. ${demandaLideresTotal > 0 ? '+ ' + demandaLideresTotal + ' Líd.' : ''}`;
        document.getElementById('resumoFixo').innerText = `${fixosCozinha} Coz. + ${fixosLider} Líd.`;
        document.getElementById('tituloRelatorio').innerText = `ESCALA DE EXTRAS - ${periodoSelecionado.toUpperCase()} - ${dataFormatada}`;
        document.getElementById('custoTotal').innerText = `R$ ${custoTotal.toLocaleString('pt-BR', {minimumFractionDigits: 2})}`;
        
        const badge = document.getElementById('badgeStatus');
        if (totalHospedes > 400) {
            badge.innerText = `Alta Ocupação (${totalHospedes} Pax)`;
            badge.style.backgroundColor = "#e11d48"; 
        } else {
            badge.innerText = `Ocupação Normal (${totalHospedes} Pax)`;
            badge.style.backgroundColor = "var(--primary)"; 
        }

        document.getElementById('resultadoBox').style.display = 'block';
        document.getElementById('btnImprimir').style.display = 'inline-flex';

        if (salvarNoHistorico) {
            const novaEscala = {
                id: Date.now(), data: dataFormatada, dataOriginal: dataInput, periodo: periodoSelecionado,
                adultos: qtdAdultos, criancas: qtdCriancas, buffet: pontosBuffet,
                fixosCozinha: fixosCozinha, fixosLider: fixosLider, totalValor: custoTotal, totalHospedes: totalHospedes
            };
            let historico = JSON.parse(localStorage.getItem('historicoEscalas')) || [];
            historico = historico.filter(item => !(item.data === novaEscala.data && item.periodo === novaEscala.periodo));
            historico.push(novaEscala);
            localStorage.setItem('historicoEscalas', JSON.stringify(historico));
            renderizarHistorico();
        }
    }

    function renderizarHistorico() {
        const historico = JSON.parse(localStorage.getItem('historicoEscalas')) || [];
        const historicoCorpo = document.getElementById('historicoCorpo');
        historicoCorpo.innerHTML = "";

        if (historico.length === 0) {
            historicoCorpo.innerHTML = `<tr><td colspan="5" style="text-align: center; color: var(--text-muted); padding: 20px;">Nenhuma escala gravada no histórico ainda.</td></tr>`;
            return;
        }

        historico.sort((a, b) => b.id - a.id).forEach(item => {
            const tr = document.createElement('tr');
            tr.innerHTML = `
                <td><strong>${item.data}</strong></td>
                <td>${item.periodo}</td>
                <td style="text-align: center;">${item.totalHospedes} Pax</td>
                <td style="text-align: right; font-weight: bold; color: var(--primary)">R$ ${item.totalValor.toLocaleString('pt-BR', {minimumFractionDigits: 2})}</td>
                <td style="text-align: center; display: flex; gap: 8px; justify-content: center;">
                    <button class="btn-ver btn-acao-hist" onclick="restaurarEscala(${item.id})">🔍 Ver</button>
                    <button class="btn-del btn-acao-hist" onclick="deletarHistorico(${item.id})">🗑️</button>
                </td>
            `;
            historicoCorpo.appendChild(tr);
        });
    }

    function restaurarEscala(id) {
        const historico = JSON.parse(localStorage.getItem('historicoEscalas')) || [];
        const escala = historico.find(item => item.id === id);

        if (escala) {
            document.getElementById('dataEscala').value = escala.dataOriginal;
            document.getElementById('periodo').value = escala.periodo;
            document.getElementById('adultos').value = escala.adultos;
            document.getElementById('criancas').value = escala.criancas;
            
            const buffetSelect = document.getElementById('buffet');
            buffetSelect.disabled = true;
            buffetSelect.value = escala.buffet;

            document.getElementById('fixosCozinha').value = escala.fixosCozinha;
            document.getElementById('fixosLider').value = escala.fixosLider;

            calcularEscala(false);
            document.getElementById('resultadoBox').scrollIntoView({ behavior: 'smooth' });
        }
    }

    function deletarHistorico(id) {
        if(confirm("Tem certeza que deseja excluir esta escala do histórico?")) {
            let historico = JSON.parse(localStorage.getItem('historicoEscalas')) || [];
            historico = historico.filter(item => item.id !== id);
            localStorage.setItem('historicoEscalas', JSON.stringify(historico));
            renderizarHistorico();
        }
    }

    function exportarBackup() {
        const historico = JSON.parse(localStorage.getItem('historicoEscalas')) || [];
        if(historico.length === 0) { alert("Não há dados para exportar."); return; }
        let csvContent = "\uFEFFID,DataFormatada,DataOriginal,Periodo,Adultos,Criancas,Buffet,FixosCozinha,FixosLider,TotalValor,TotalHospedes\n";
        historico.forEach(item => {
            csvContent += `${item.id},${item.data},${item.dataOriginal},"${item.periodo}",${item.adultos},${item.criancas},${item.buffet},${item.fixosCozinha},${item.fixosLider},${item.totalValor},${item.totalHospedes}\n`;
        });
        const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
        const url = URL.createObjectURL(blob);
        const link = document.createElement("a");
        link.setAttribute("href", url);
        link.setAttribute("download", `backup_escalas_bourbon_${new Date().toISOString().slice(0,10)}.csv`);
        document.body.appendChild(link); link.click(); document.body.removeChild(link);
    }

    function importarBackup(event) {
        const file = event.target.files[0]; if (!file) return;
        const reader = new FileReader();
        reader.onload = function(e) {
            const rows = e.target.result.split('\n');
            let historicoImportado = [];
            for (let i = 1; i < rows.length; i++) {
                if (!rows[i].trim()) continue;
                const cols = rows[i].split(/,(?=(?:(?:[^"]*"){2})*[^"]*$)/);
                if (cols.length >= 11) {
                    historicoImportado.push({
                        id: parseInt(cols[0]), data: cols[1], dataOriginal: cols[2], periodo: cols[3].replace(/"/g, ''),
                        adultos: parseInt(cols[4]), criancas: parseInt(cols[5]), buffet: cols[6],
                        fixosCozinha: parseInt(cols[7]), fixosLider: parseInt(cols[8]), totalValor: parseFloat(cols[9]), totalHospedes: parseInt(cols[10])
                    });
                }
            }
            if (historicoImportado.length > 0 && confirm(`Deseja importar ${historicoImportado.length} registros?`)) {
                let atual = JSON.parse(localStorage.getItem('historicoEscalas')) || [];
                historicoImportado.forEach(n => { atual = atual.filter(i => i.id !== n.id); atual.push(n); });
                localStorage.setItem('historicoEscalas', JSON.stringify(atual)); renderizarHistorico();
                alert("Banco de dados atualizado!");
            }
        };
        reader.readAsText(file); event.target.value = '';
    }
