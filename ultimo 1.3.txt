<script>
    let pedidoAtual = null;

    document.getElementById('dataEscala').valueAsDate = new Date();
    window.onload = renderizarHistorico;

    function alternarAba(modo) {
        if (modo === 'detalhado') {
            document.body.classList.add('modo-detalhado');
            document.getElementById('btnTabDetalhado').classList.add('active');
            document.getElementById('btnTabResumo').classList.remove('active');
        } else {
            document.body.classList.remove('modo-detalhado');
            document.getElementById('btnTabResumo').classList.add('active');
            document.getElementById('btnTabDetalhado').classList.remove('active');
        }
    }

    function gerarPedido() {
        const dataInput = document.getElementById('dataEscala').value;
        const turno = document.getElementById('periodo').value;

        if (!dataInput) {
            alert("Selecione uma data!");
            return;
        }

        const partes = dataInput.split('-');
        const dataFormatada = `${partes[2]}/${partes[1]}/${partes[0]}`;
        document.getElementById('exibicaoData').innerText = `📅 Data da Operação: ${dataFormatada}`;

        // Exemplo de dados calculados
        const itens = [
            { local: "Restaurante TOM", turno: turno, cargo: "Cozinheiro", qtd: 6, entrada: "06:00", valor: 180, subtotal: 1080 },
            { local: "Restaurante TOM", turno: turno, cargo: "Steward", qtd: 4, entrada: "06:00", valor: 140, subtotal: 560 },
            { local: "Restaurante KIBÔ", turno: turno, cargo: "Cozinheiro", qtd: 5, entrada: "16:00", valor: 180, subtotal: 900 },
            { local: "Restaurante KIBÔ", turno: turno, cargo: "Steward", qtd: 5, entrada: "17:00", valor: 140, subtotal: 700 }
        ];

        pedidoAtual = {
            data: dataFormatada,
            dataOriginal: dataInput,
            turno: turno,
            itens: itens
        };

        renderizarTabela();
        document.getElementById('resultadoBox').style.display = 'block';
    }

    function renderizarTabela() {
        const tbody = document.getElementById('tabelaCorpo');
        tbody.innerHTML = "";

        let totalQtd = 0;
        let totalCusto = 0;

        pedidoAtual.itens.forEach(item => {
            totalQtd += item.qtd;
            totalCusto += item.subtotal;

            const tr = document.createElement('tr');
            tr.innerHTML = `
                <td class="col-local">${item.local}</td>
                <td>${item.turno}</td>
                <td><strong>${item.cargo}</strong></td>
                <td style="text-align: center;"><strong>${item.qtd}</strong></td>
                <td>${item.entrada}</td>
                <td style="text-align: right;">R$ ${item.valor.toFixed(2)}</td>
                <td style="text-align: right;"><strong>R$ ${item.subtotal.toFixed(2)}</strong></td>
            `;
            tbody.appendChild(tr);
        });

        document.getElementById('totalPessoas').innerText = `Total de Pessoas Extras: ${totalQtd}`;
        document.getElementById('custoGeral').innerText = `Custo Geral Estimado: R$ ${totalCusto.toLocaleString('pt-BR', {minimumFractionDigits: 2})}`;
    }

    /* Impressão em Página Única */
    function imprimir(tipo) {
        document.body.classList.remove('print-resumo', 'print-detalhado');
        document.body.classList.add(`print-${tipo}`);

        window.print();

        setTimeout(() => {
            document.body.classList.remove(`print-${tipo}`);
        }, 1000);
    }

    /* Exportação Excel contendo a Data */
    function exportarExcel() {
        if (!pedidoAtual) return;

        const dadosExcel = pedidoAtual.itens.map(i => ({
            'Data Operação': pedidoAtual.data,
            'Local / Setor': i.local,
            'Turno': i.turno,
            'Cargo / Função': i.cargo,
            'Qtd Extra': i.qtd,
            'Entrada': i.entrada,
            'Valor Diária (R$)': i.valor,
            'Subtotal (R$)': i.subtotal
        }));

        const wb = XLSX.utils.book_new();
        const ws = XLSX.utils.json_to_sheet(dadosExcel);
        XLSX.utils.book_append_sheet(wb, ws, "Pedido Extras");
        XLSX.writeFile(wb, `Pedido_Extras_${pedidoAtual.data.replace(/\//g, '-')}.xlsx`);
    }

    function salvarHistorico() {
        let historico = JSON.parse(localStorage.getItem('historicoPedidos')) || [];
        const novoItem = {
            id: Date.now(),
            data: pedidoAtual.data,
            dataOriginal: pedidoAtual.dataOriginal,
            turno: pedidoAtual.turno,
            qtd: pedidoAtual.itens.reduce((a, b) => a + b.qtd, 0),
            custo: pedidoAtual.itens.reduce((a, b) => a + b.subtotal, 0),
            itens: pedidoAtual.itens
        };

        historico.unshift(novoItem);
        localStorage.setItem('historicoPedidos', JSON.stringify(historico));
        renderizarHistorico();
        alert("Pedido gravado com sucesso!");
    }

    function renderizarHistorico() {
        const historico = JSON.parse(localStorage.getItem('historicoPedidos')) || [];
        const tbody = document.getElementById('historicoCorpo');
        tbody.innerHTML = "";

        if (historico.length === 0) {
            tbody.innerHTML = `<tr><td colspan="5" style="text-align: center; color: var(--text-muted); padding: 15px;">Nenhum pedido gravado.</td></tr>`;
            return;
        }

        historico.forEach(item => {
            const tr = document.createElement('tr');
            tr.innerHTML = `
                <td><strong>${item.data}</strong></td>
                <td>${item.turno}</td>
                <td>${item.qtd} pax</td>
                <td><strong>R$ ${item.custo.toLocaleString('pt-BR', {minimumFractionDigits: 2})}</strong></td>
                <td style="text-align: center; display: flex; gap: 8px; justify-content: center;">
                    <button class="btn-ver" onclick="restaurar(${item.id})">👁️ Ver</button>
                    <button class="btn-del" onclick="excluirComLog(${item.id})">🗑️ Excluir</button>
                </td>
            `;
            tbody.appendChild(tr);
        });
    }

    function restaurar(id) {
        const historico = JSON.parse(localStorage.getItem('historicoPedidos')) || [];
        const item = historico.find(i => i.id === id);
        if (item) {
            pedidoAtual = item;
            document.getElementById('exibicaoData').innerText = `📅 Data da Operação: ${item.data}`;
            renderizarTabela();
            document.getElementById('resultadoBox').style.display = 'block';
            document.getElementById('resultadoBox').scrollIntoView({ behavior: 'smooth' });
        }
    }

    /* Exclusão com Registro do Usuário (Opção B - Log de Auditoria) */
    function excluirComLog(id) {
        const usuario = prompt("Para confirmar a exclusão, digite seu NOME DE USUÁRIO:");
        if (!usuario || !usuario.trim()) {
            alert("Ação cancelada! É obrigatório informar o usuário para o log de auditoria.");
            return;
        }

        let historico = JSON.parse(localStorage.getItem('historicoPedidos')) || [];
        const item = historico.find(i => i.id === id);

        if (item) {
            // Salva o Log na tabela de auditoria
            let logs = JSON.parse(localStorage.getItem('auditLogsExclusao')) || [];
            logs.push({
                acao: 'EXCLUSAO_PEDIDO',
                pedidoId: id,
                dataOperacao: item.data,
                usuarioExcluiu: usuario.trim(),
                dataHoraExclusao: new Date().toLocaleString('pt-BR')
            });
            localStorage.setItem('auditLogsExclusao', JSON.stringify(logs));

            // Remove o registro
            historico = historico.filter(i => i.id !== id);
            localStorage.setItem('historicoPedidos', JSON.stringify(historico));
            renderizarHistorico();

            alert(`Registro #${id} excluído com sucesso por ${usuario.trim()}. O evento foi registrado no Log de Auditoria.`);
        }
    }
</script>
