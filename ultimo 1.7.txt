    function inicializarDadosSemana() {
        const hoje = new Date();
        const diasNomes = ["Domingo", "Segunda", "Terça", "Quarta", "Quinta", "Sexta", "Sábado"];

        dadosSemana = [];
        for (let i = 0; i < 7; i++) {
            let dataDia = new Date(hoje);
            dataDia.setDate(hoje.getDate() + i);

            let paxGlobal = {
                manha: { adultos: 500, criancas: 0 },
                tarde: { adultos: 350, criancas: 0 },
                madrugada: { adultos: 350, criancas: 0 }
            };

            let estruturaSetores = {};
            const mockDia = { paxGlobal };

            LISTA_SETORES.forEach(itemSetor => {
                const turnosIniciais = { manha: true, tarde: true, madrugada: false };
                let setorObj = {
                    tipo: itemSetor.tipo,
                    operando: true,
                    turnosAtivos: turnosIniciais,
                    definido: false,
                    paxReserva: {
                        ativo: false,
                        manha: 500,
                        tarde: 350,
                        madrugada: 350
                    }
                };
                setorObj.cargos = calcularCargosPorSetor(setorObj, mockDia);
                estruturaSetores[itemSetor.nome] = setorObj;
            });

            const ano = dataDia.getFullYear();
            const mes = String(dataDia.getMonth() + 1).padStart(2, '0');
            const dia = String(dataDia.getDate()).padStart(2, '0');

            dadosSemana.push({
                diaNome: diasNomes[dataDia.getDay()],
                data: `${ano}-${mes}-${dia}`,
                paxGlobal: paxGlobal,
                setores: estruturaSetores
            });
        }
    }

    function renderizarAbas() {
        const container = document.getElementById('containerAbasSemana');
        container.innerHTML = "";
        dadosSemana.forEach((dia, index) => {
            const totalPaxCafes = dia.paxGlobal.manha.adultos + dia.paxGlobal.manha.criancas;
            const btn = document.createElement('div');
            btn.className = `aba-dia ${index === diaAtivoIndex ? 'ativa' : ''}`;
            btn.onclick = () => trocardia(index);
            btn.innerHTML = `
                <span class="dia-nome">${dia.diaNome}</span>
                <span class="dia-pax">${totalPaxCafes} PAX (Café)</span>
            `;
            container.appendChild(btn);
        });
    }

    function trocardia(index) {
        diaAtivoIndex = index;
        document.getElementById('secaoPedidoValorizado').style.display = 'none';
        renderizarAbas();
        carregarDiaAtivo();
    }

    function carregarDiaAtivo() {
        const dia = dadosSemana[diaAtivoIndex];
        document.getElementById('dataEscala').value = dia.data;
        
        document.getElementById('paxManhaAd').value = dia.paxGlobal.manha.adultos;
        document.getElementById('paxManhaCr').value = dia.paxGlobal.manha.criancas;
        document.getElementById('lblTotalPaxManha').innerText = `${dia.paxGlobal.manha.adultos + dia.paxGlobal.manha.criancas} PAX`;

        document.getElementById('paxTardeCr').value = dia.paxGlobal.tarde.criancas;
        document.getElementById('lblTotalPaxTarde').innerText = `${dia.paxGlobal.tarde.adultos + dia.paxGlobal.tarde.criancas} PAX`;

        document.getElementById('paxMadrugadaAd').value = dia.paxGlobal.madrugada.adultos;
        document.getElementById('paxMadrugadaCr').value = dia.paxGlobal.madrugada.criancas;
        document.getElementById('lblTotalPaxMadrugada').innerText = `${dia.paxGlobal.madrugada.adultos + dia.paxGlobal.madrugada.criancas} PAX`;

    function atualizarDataDiaAtual() {
        const dataVal = document.getElementById('dataEscala').value;
        if (!dataVal) return;

        const partes = dataVal.split('-');
        const diasNomes = ["Domingo", "Segunda", "Terça", "Quarta", "Quinta", "Sexta", "Sábado"];

        for (let i = 0; i < dadosSemana.length; i++) {
            if (i >= diaAtivoIndex) {
                let dataDia = new Date(partes[0], partes[1] - 1, partes[2]);
                dataDia.setDate(dataDia.getDate() + (i - diaAtivoIndex));

                const ano = dataDia.getFullYear();
                const mes = String(dataDia.getMonth() + 1).padStart(2, '0');
                const dia = String(dataDia.getDate()).padStart(2, '0');

                dadosSemana[i].data = `${ano}-${mes}-${dia}`;
                dadosSemana[i].diaNome = diasNomes[dataDia.getDay()];
            }
        }

        renderizarAbas();
    }

    function atualizarPaxGlobal(replicarTardeParaMadrugada = false) {
        const diaAtivo = dadosSemana[diaAtivoIndex];

        diaAtivo.paxGlobal.manha.adultos = parseInt(document.getElementById('paxManhaAd').value) || 0;
        diaAtivo.paxGlobal.manha.criancas = parseInt(document.getElementById('paxManhaCr').value) || 0;

        diaAtivo.paxGlobal.tarde.adultos = parseInt(document.getElementById('paxTardeAd').value) || 0;
        diaAtivo.paxGlobal.tarde.criancas = parseInt(document.getElementById('paxTardeCr').value) || 0;

        if (replicarTardeParaMadrugada) {
            diaAtivo.paxGlobal.madrugada.adultos = diaAtivo.paxGlobal.tarde.adultos;
            diaAtivo.paxGlobal.madrugada.criancas = diaAtivo.paxGlobal.tarde.criancas;
            document.getElementById('paxMadrugadaAd').value = diaAtivo.paxGlobal.madrugada.adultos;
            document.getElementById('paxMadrugadaCr').value = diaAtivo.paxGlobal.madrugada.criancas;
        } else {
            diaAtivo.paxGlobal.madrugada.adultos = parseInt(document.getElementById('paxMadrugadaAd').value) || 0;
            diaAtivo.paxGlobal.madrugada.criancas = parseInt(document.getElementById('paxMadrugadaCr').value) || 0;
        }

        document.getElementById('lblTotalPaxManha').innerText = `${diaAtivo.paxGlobal.manha.adultos + diaAtivo.paxGlobal.manha.criancas} PAX`;
        document.getElementById('lblTotalPaxTarde').innerText = `${diaAtivo.paxGlobal.tarde.adultos + diaAtivo.paxGlobal.tarde.criancas} PAX`;
        document.getElementById('lblTotalPaxMadrugada').innerText = `${diaAtivo.paxGlobal.madrugada.adultos + diaAtivo.paxGlobal.madrugada.criancas} PAX`;

        recalcularCargosDosSetores();
        renderizarAbas();
        renderizarSetoresDiaAtivo();
        verificarEValidadorGeral();

        if (document.getElementById('secaoPedidoValorizado').style.display === 'block') {
            abrirPedidoValorizado();
        }
    }

    function recalcularCargosDosSetores() {
        const diaAtivo = dadosSemana[diaAtivoIndex];
        Object.keys(diaAtivo.setores).forEach(nomeSetor => {
            const setor = diaAtivo.setores[nomeSetor];
            if(setor.operando) {
                const calculados = calcularCargosPorSetor(setor, diaAtivo);
                ['manha', 'tarde', 'madrugada'].forEach(turnoKey => {
                    if (setor.turnosAtivos[turnoKey]) {
                        calculados[turnoKey].forEach((novoCargo, idx) => {
                            if (setor.cargos[turnoKey] && setor.cargos[turnoKey][idx]) {
                                setor.cargos[turnoKey][idx].exigido = novoCargo.exigido;
                            } else {
                                if(!setor.cargos[turnoKey]) setor.cargos[turnoKey] = [];
                                setor.cargos[turnoKey].push(novoCargo);
                            }
                        });
                    }
                });
            }
        });
    }

    function toggleAbrirSetor(nomeSetor) {
        setoresAbertos[nomeSetor] = !setoresAbertos[nomeSetor];
        const el = document.getElementById(`corpo-${sanitizarId(nomeSetor)}`);
        if (el) el.style.display = setoresAbertos[nomeSetor] ? 'block' : 'none';
    }

    function sanitizarId(str) {
        return str.replace(/[^a-zA-Z0-9]/g, '_');
    }

    function alterarStatusOperacao(nomeSetor, statusOperando) {
        const diaAtivo = dadosSemana[diaAtivoIndex];
        const setor = diaAtivo.setores[nomeSetor];
        setor.operando = statusOperando;
        setor.definido = true;

        if (setor.operando) {
            setor.cargos = calcularCargosPorSetor(setor, diaAtivo);
        } else {
            setor.cargos = { manha: [], tarde: [], madrugada: [] };
        }

        renderizarSetoresDiaAtivo();
        verificarEValidadorGeral();

        if (document.getElementById('secaoPedidoValorizado').style.display === 'block') {
            abrirPedidoValorizado();
        }
    }

    function toggleTurnoSetor(nomeSetor, turnoKey, checkbox) {
        const diaAtivo = dadosSemana[diaAtivoIndex];
        const setor = diaAtivo.setores[nomeSetor];
        setor.turnosAtivos[turnoKey] = checkbox.checked;

        if (checkbox.checked) {
            const rec = calcularCargosPorSetor(setor, diaAtivo);
            setor.cargos[turnoKey] = rec[turnoKey];
        } else {
            setor.cargos[turnoKey] = [];
        }

        setor.definido = true;
        renderizarSetoresDiaAtivo();
        verificarEValidadorGeral();

        if (document.getElementById('secaoPedidoValorizado').style.display === 'block') {
            abrirPedidoValorizado();
        }
    }

    function togglePaxReserva(nomeSetor, checkbox) {
        const diaAtivo = dadosSemana[diaAtivoIndex];
        const setor = diaAtivo.setores[nomeSetor];
        setor.paxReserva.ativo = checkbox.checked;

        recalcularCargosDosSetores();
        renderizarSetoresDiaAtivo();
        verificarEValidadorGeral();

        if (document.getElementById('secaoPedidoValorizado').style.display === 'block') {
            abrirPedidoValorizado();
        }
    }

    function atualizarPaxReservaInput(nomeSetor, turnoKey, valor) {
        const diaAtivo = dadosSemana[diaAtivoIndex];
        const setor = diaAtivo.setores[nomeSetor];

    function renderizarLinhasCargo(nomeSetor, turnoKey) {
        const diaAtivo = dadosSemana[diaAtivoIndex];
        const cargos = diaAtivo.setores[nomeSetor].cargos[turnoKey] || [];
        if (cargos.length === 0) return `<p style="font-size:12px; color:var(--text-muted);">Nenhuma função parametrizada.</p>`;

        let html = "";
        cargos.forEach((cargo, index) => {
            const disponivelFixo = cargo.fixo || 0;
            const exigido = cargo.exigido || 0;

            const somaFracionamentos = (cargo.fracionamentos || []).reduce((acc, f) => acc + (parseInt(f.qtd) || 0), 0);
            const extraPrincipal = Math.max(0, exigido - disponivelFixo - somaFracionamentos);

            html += `
                <div class="grid-funcao-linha">
                    <div class="campo">
                        <label>Função</label>
                        <select onchange="atualizarCargoProp('${nomeSetor}', '${turnoKey}', ${index}, 'nome', this.value)">
                            ${Object.keys(VALORES_PADRAO_DIARIA).map(f => `<option value="${f}" ${f === cargo.nome ? 'selected' : ''}>${f}</option>`).join('')}
                        </select>
                    </div>
                    <div class="campo">
                        <label>Entrada</label>
                        <select onchange="atualizarCargoProp('${nomeSetor}', '${turnoKey}', ${index}, 'entrada', this.value)">
                            ${HORARIOS_POR_TURNO[turnoKey].map(h => `<option value="${h}" ${h === cargo.entrada ? 'selected' : ''}>${h}</option>`).join('')}
                        </select>
                    </div>
                    <div class="campo">
                        <label>Exigido</label>
                        <div class="badge-qtd-exigido">${exigido}</div>
                    </div>
                    <div class="campo">
                        <label>Quadro Fixo</label>
                        <input type="number" value="${disponivelFixo}" min="0" onchange="atualizarCargoProp('${nomeSetor}', '${turnoKey}', ${index}, 'fixo', this.value)">
                    </div>
                    <div class="campo">
                        <label>Extras</label>
                        <div class="badge-qtd-extra">${extraPrincipal}</div>
                    </div>
                    <button class="btn-acao-linha btn-add-fracao" title="Fracionar Horário de Entrada" onclick="adicionarFracionamento('${nomeSetor}', '${turnoKey}', ${index})">⏱️</button>
                    <button class="btn-acao-linha btn-del-linha" title="Excluir Linha" onclick="removerCargo('${nomeSetor}', '${turnoKey}', ${index})">🗑️</button>
                </div>
            `;

            if (cargo.fracionamentos && cargo.fracionamentos.length > 0) {
                cargo.fracionamentos.forEach((frac, fIdx) => {
                    html += `
                        <div class="grid-funcao-linha fracionada">
                            <div style="font-size:11px; font-weight:bold; color:var(--primary);">↳ Fracionamento ${fIdx + 1}</div>
                            <div class="campo">
                                <label>Entrada</label>
                                <select onchange="atualizarFracaoProp('${nomeSetor}', '${turnoKey}', ${index}, ${fIdx}, 'entrada', this.value)">
                                    ${HORARIOS_POR_TURNO[turnoKey].map(h => `<option value="${h}" ${h === frac.entrada ? 'selected' : ''}>${h}</option>`).join('')}
                                </select>
                            </div>
                            <div class="campo">
                                <label>Qtd Extras</label>
                                <input type="number" value="${frac.qtd}" min="1" onchange="atualizarFracaoProp('${nomeSetor}', '${turnoKey}', ${index}, ${fIdx}, 'qtd', this.value)">
                            </div>
                            <div colspan="2"></div>
                            <button class="btn-acao-linha btn-del-linha" title="Remover Fracionamento" onclick="removerFracionamento('${nomeSetor}', '${turnoKey}', ${index}, ${fIdx})">✕</button>
                        </div>
                    `;
                });
            }
        });

        return html;
    }

    function atualizarCargoProp(nomeSetor, turnoKey, index, prop, valor) {
        const diaAtivo = dadosSemana[diaAtivoIndex];
        const cargo = diaAtivo.setores[nomeSetor].cargos[turnoKey][index];
        cargo[prop] = prop === 'exigido' || prop === 'fixo' ? (parseInt(valor) || 0) : valor;
        diaAtivo.setores[nomeSetor].definido = true;
        renderizarSetoresDiaAtivo();
        verificarEValidadorGeral();

        if (document.getElementById('secaoPedidoValorizado').style.display === 'block') {
            abrirPedidoValorizado();
        }
    }

    function adicionarNovoCargo(nomeSetor, turnoKey) {

            fixo: 0,

        };

            abrirPedidoValorizado();
        }
    }

    function removerCargo(nomeSetor, turnoKey, index) {
        const diaAtivo = dadosSemana[diaAtivoIndex];
        diaAtivo.setores[nomeSetor].cargos[turnoKey].splice(index, 1);
        diaAtivo.setores[nomeSetor].definido = true;
        renderizarSetoresDiaAtivo();
        verificarEValidadorGeral();

        if (document.getElementById('secaoPedidoValorizado').style.display === 'block') {
            abrirPedidoValorizado();
        }
    }

    function adicionarFracionamento(nomeSetor, turnoKey, cargoIndex) {
        const diaAtivo = dadosSemana[diaAtivoIndex];
        const cargo = diaAtivo.setores[nomeSetor].cargos[turnoKey][cargoIndex];
        if (!cargo.fracionamentos) cargo.fracionamentos = [];
        cargo.fracionamentos.push({
            entrada: HORARIOS_POR_TURNO[turnoKey][1] || HORARIOS_POR_TURNO[turnoKey][0],
            qtd: 1
        });
        diaAtivo.setores[nomeSetor].definido = true;
        renderizarSetoresDiaAtivo();
        verificarEValidadorGeral();

        if (document.getElementById('secaoPedidoValorizado').style.display === 'block') {
            abrirPedidoValorizado();
        }
    }

    function atualizarFracaoProp(nomeSetor, turnoKey, cargoIndex, fIdx, prop, valor) {
        const diaAtivo = dadosSemana[diaAtivoIndex];
        const frac = diaAtivo.setores[nomeSetor].cargos[turnoKey][cargoIndex].fracionamentos[fIdx];
        frac[prop] = prop === 'qtd' ? (parseInt(valor) || 1) : valor;
        diaAtivo.setores[nomeSetor].definido = true;
        renderizarSetoresDiaAtivo();
        verificarEValidadorGeral();

        if (document.getElementById('secaoPedidoValorizado').style.display === 'block') {
            abrirPedidoValorizado();
        }
    }

    function removerFracionamento(nomeSetor, turnoKey, cargoIndex, fIdx) {
        const diaAtivo = dadosSemana[diaAtivoIndex];
        diaAtivo.setores[nomeSetor].cargos[turnoKey][cargoIndex].fracionamentos.splice(fIdx, 1);
        diaAtivo.setores[nomeSetor].definido = true;
        renderizarSetoresDiaAtivo();
        verificarEValidadorGeral();

        if (document.getElementById('secaoPedidoValorizado').style.display === 'block') {
            abrirPedidoValorizado();
        }
    }

    function verificarEValidadorGeral() {
        const diaAtivo = dadosSemana[diaAtivoIndex];
        let todosDefinidos = true;

        Object.keys(diaAtivo.setores).forEach(nomeSetor => {
            if (!diaAtivo.setores[nomeSetor].definido) {
                todosDefinidos = false;
            }
        });

        const btn = document.getElementById('btnGerarRelatorio');
        if (todosDefinidos) {
            btn.disabled = false;
            btn.className = "btn-gerar-relatorio liberado";
            btn.innerHTML = "🔓 Liberar Pedido Valorizado do Dia";
        } else {
            btn.disabled = true;
            btn.className = "btn-gerar-relatorio";
            btn.innerHTML = "🔒 Sinalize todos os locais para liberar o pedido";
        }
    }

    function alternarSubTab(tab) {
        document.getElementById('subTabResumo').className = `btn-subtab ${tab === 'resumo' ? 'ativa' : ''}`;
        document.getElementById('subTabDetalhamento').className = `btn-subtab ${tab === 'detalhamento' ? 'ativa' : ''}`;
        
        document.getElementById('conteudoSubTabResumo').style.display = tab === 'resumo' ? 'block' : 'none';
        document.getElementById('conteudoSubTabDetalhamento').style.display = tab === 'detalhamento' ? 'block' : 'none';
    }

    function abrirPedidoValorizado() {
        const diaAtivo = dadosSemana[diaAtivoIndex];
        const resumoMap = {};
        const detalhamentoList = [];

        let totalQtdExtras = 0;
        let totalCustoGeral = 0;

        Object.keys(diaAtivo.setores).forEach(nomeSetor => {
            const setor = diaAtivo.setores[nomeSetor];
            if (setor.operando) {
                ['manha', 'tarde', 'madrugada'].forEach(turnoKey => {
                    if (setor.turnosAtivos[turnoKey] && setor.cargos[turnoKey]) {
                        setor.cargos[turnoKey].forEach(cargo => {
                            const valorDiaria = VALORES_PADRAO_DIARIA[cargo.nome] || 150.00;
                            const somaFracionamentos = (cargo.fracionamentos || []).reduce((acc, f) => acc + (parseInt(f.qtd) || 0), 0);
                            const extraBase = Math.max(0, cargo.exigido - cargo.fixo - somaFracionamentos);

                            if (extraBase > 0) {
                                totalQtdExtras += extraBase;
                                const sub = extraBase * valorDiaria;
                                totalCustoGeral += sub;

                                detalhamentoList.push({
                                    setor: nomeSetor,
                                    turno: NOME_TURNO_LABEL[turnoKey],
                                    cargo: cargo.nome,
                                    qtd: extraBase,
                                    entrada: cargo.entrada,
                                    valorDiaria,
                                    subtotal: sub
                                });

                                const keyRes = `${cargo.nome}_${turnoKey}`;
                                if (!resumoMap[keyRes]) {
                                    resumoMap[keyRes] = {
                                        cargo: cargo.nome,
                                        turno: NOME_TURNO_LABEL[turnoKey],
                                        horariosContagem: {},
                                        qtd: 0,
                                        valorDiaria,
                                        subtotal: 0
                                    };
                                }
                                resumoMap[keyRes].qtd += extraBase;
                                resumoMap[keyRes].horariosContagem[cargo.entrada] = (resumoMap[keyRes].horariosContagem[cargo.entrada] || 0) + extraBase;
                                resumoMap[keyRes].subtotal += sub;
                            }

                            if (cargo.fracionamentos) {
                                cargo.fracionamentos.forEach(frac => {
                                    if (frac.qtd > 0) {
                                        totalQtdExtras += frac.qtd;
                                        const subFrac = frac.qtd * valorDiaria;
                                        totalCustoGeral += subFrac;

                                        detalhamentoList.push({
                                            setor: nomeSetor,
                                            turno: NOME_TURNO_LABEL[turnoKey],
                                            cargo: cargo.nome,
                                            qtd: frac.qtd,
                                            entrada: frac.entrada,
                                            valorDiaria,
                                            subtotal: subFrac
                                        });

                                        const keyRes = `${cargo.nome}_${turnoKey}`;
                                        if (!resumoMap[keyRes]) {
                                            resumoMap[keyRes] = {
                                                cargo: cargo.nome,
                                                turno: NOME_TURNO_LABEL[turnoKey],
                                                horariosContagem: {},
                                                qtd: 0,
                                                valorDiaria,
                                                subtotal: 0
                                            };
                                        }
                                        resumoMap[keyRes].qtd += frac.qtd;
                                        resumoMap[keyRes].horariosContagem[frac.entrada] = (resumoMap[keyRes].horariosContagem[frac.entrada] || 0) + frac.qtd;
                                        resumoMap[keyRes].subtotal += subFrac;
                                    }
                                });
                            }
                        });
                    }
                });
            }
        });

        const listaResumoFinal = Object.values(resumoMap).map(item => {
            const horariosFormatados = Object.entries(item.horariosContagem)
                .sort(([h1], [h2]) => h1.localeCompare(h2))
                .map(([horario, q]) => `${horario} (${q}x)`)
                .join(', ');
            return {
                cargo: item.cargo,
                turno: item.turno,
                horariosTexto: horariosFormatados,
                qtd: item.qtd,
                valorDiaria: item.valorDiaria,
                subtotal: item.subtotal
            };
        });

        relatorioAtualDados = {
            resumo: listaResumoFinal,
            detalhamento: detalhamentoList,
            totalQtdExtras,
            totalCustoGeral
        };

        renderizarTabelasRelatorio(relatorioAtualDados);

        document.getElementById('secaoPedidoValorizado').style.display = 'block';
        document.getElementById('secaoPedidoValorizado').scrollIntoView({ behavior: 'smooth' });
    }

    function renderizarTabelasRelatorio(dados) {
        const tbodyRes = document.getElementById('corpoTabelaValores');
        tbodyRes.innerHTML = "";
        dados.resumo.forEach(item => {
            const tr = document.createElement('tr');
            tr.innerHTML = `
                <td><strong>${item.cargo}</strong></td>
                <td>${item.turno}</td>
                <td>${item.horariosTexto}</td>
                <td><strong>${item.qtd}</strong></td>
                <td>R$ ${item.valorDiaria.toFixed(2)}</td>
                <td><strong>R$ ${item.subtotal.toFixed(2)}</strong></td>
            `;
            tbodyRes.appendChild(tr);
        });

        const tbodyDet = document.getElementById('corpoTabelaDetalhamento');
        tbodyDet.innerHTML = "";
        dados.detalhamento.forEach(item => {
            const tr = document.createElement('tr');
            tr.innerHTML = `
                <td>${item.setor}</td>
                <td>${item.turno}</td>
                <td><strong>${item.cargo}</strong></td>
                <td><strong>${item.qtd}</strong></td>
                <td>${item.entrada}</td>
                <td>R$ ${item.valorDiaria.toFixed(2)}</td>
                <td><strong>R$ ${item.subtotal.toFixed(2)}</strong></td>
            `;
            tbodyDet.appendChild(tr);
        });

        document.getElementById('lblTotalExtrasQtde').innerText = dados.totalQtdExtras;
        document.getElementById('lblTotalCustoReais').innerText = dados.totalCustoGeral.toLocaleString('pt-BR', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    }

    function gravarPedidoAtual() {
        if (!relatorioAtualDados.resumo || relatorioAtualDados.resumo.length === 0) {
            alert('Gere primeiro o pedido antes de gravar!');
            return;
        }

        const diaAtivo = dadosSemana[diaAtivoIndex];
        const pedidos = JSON.parse(localStorage.getItem('pedidos_bourbon') || '[]');
        
        const novopedido = {
            id: Date.now(),
            data: diaAtivo.data,
            diaNome: diaAtivo.diaNome,
            qtdExtras: relatorioAtualDados.totalQtdExtras,
            custoTotal: relatorioAtualDados.totalCustoGeral,
            resumo: relatorioAtualDados.resumo,
            detalhamento: relatorioAtualDados.detalhamento
        };

        pedidos.push(novopedido);
        localStorage.setItem('pedidos_bourbon', JSON.stringify(pedidos));
        alert('Pedido gravado com sucesso no Histórico!');
    }

    function abrirModalHistorico() {
        const modal = document.getElementById('modalHistorico');
        const container = document.getElementById('conteudoHistoricoModal');
        const pedidos = JSON.parse(localStorage.getItem('pedidos_bourbon') || '[]');

        if (pedidos.length === 0) {
            container.innerHTML = `<p style="text-align:center; color:var(--text-muted); padding:20px;">Nenhum pedido gravado anteriormente.</p>`;
        } else {
            let html = `
                <table class="tabela-custom">
                    <thead>
                        <tr>
                            <th>Data Operação</th>
                            <th>Dia</th>
                            <th>Qtd Extras</th>
                            <th>Custo Total</th>
                            <th>Ações</th>
                        </tr>
                    </thead>
                    <tbody>
            `;
            pedidos.forEach((p, index) => {
                const valorFormatado = typeof p.custoTotal === 'number' 
                    ? p.custoTotal.toLocaleString('pt-BR', {minimumFractionDigits: 2}) 
                    : p.custoTotal;

                    <tr>
                        <td><strong>${p.data}</strong></td>
                        <td>${p.diaNome}</td>
                        <td>${p.qtdExtras} pax</td>
                        <td>R$ ${valorFormatado}</td>
                        <td>
                            <button class="btn-tbl-acao btn-tbl-ver" onclick="visualizarPedidoSalvo(${p.id})">👁️ Ver</button>
                            <button class="btn-tbl-acao btn-tbl-excel" onclick="exportarExcelPedidoSalvo(${p.id})">📊 Excel</button>

    }

    function visualizarPedidoSalvo(id) {
        const pedidos = JSON.parse(localStorage.getItem('pedidos_bourbon') || '[]');
        const ped = pedidos.find(p => p.id === id);
        if (!ped) return;

        if (ped.resumo && ped.detalhamento) {
            relatorioAtualDados = {
                resumo: ped.resumo,
                detalhamento: ped.detalhamento,
                totalQtdExtras: ped.qtdExtras,
                totalCustoGeral: typeof ped.custoTotal === 'number' ? ped.custoTotal : parseFloat(ped.custoTotal.replace('.', '').replace(',', '.')) || 0
            };
            renderizarTabelasRelatorio(relatorioAtualDados);
        }

        fecharModalHistorico();
        document.getElementById('secaoPedidoValorizado').style.display = 'block';
        document.getElementById('secaoPedidoValorizado').scrollIntoView({ behavior: 'smooth' });
    }

    function excluirPedidoSalvo(index) {
        if (confirm("Tem certeza que deseja excluir este registro do histórico?")) {

            pedidos.splice(index, 1);
            localStorage.setItem('pedidos_bourbon', JSON.stringify(pedidos));
            abrirModalHistorico();
        }
    }
